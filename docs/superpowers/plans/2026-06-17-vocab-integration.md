# Vocab Integration (Sub-project 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the ~28 hardcoded words with the 12.260-word enriched dictionary — a build-time normalization script produces a clean lazy-loaded dictionary plus a bundled, type-checked set of lesson nouns, and the learning path auto-extends with frequency units covering every usable noun.

**Architecture:** A Node script (`scripts/build-vocab.mjs`) reads the raw 21 MB source once on the desktop, normalizes noun gender, and emits two artifacts: `public/vocab/dictionary.json` (full rich data, lazy `fetch`) and `src/data/generated/lessonWords.ts` (minimal noun fields, in bundle). Shared types live in `src/data/vocab-types.ts`; `src/data/path.ts` appends auto-generated frequency units after Unit 25.

**Tech Stack:** Node 24 (built-in `node --test`), TypeScript ~6, Vite 6, React 19. No new dependencies.

- [ ] **Step 3: Implement the normalization library**

Create `scripts/lib/vocab-normalize.mjs`:
```js
const ARTICLES = new Set(['der', 'die', 'das']);

const GENDER_RO = {
  der: 'Masculin',
  die: 'Feminin',
  das: 'Neutru',
};

/**
 * Resolve a noun's article ('der'|'die'|'das'), or null if unrecoverable.
 * 1) clean field, 2) lowercased capitalized field, 3) declension nominative prefix.
 */
export function normalizeGender(entry) {
  const noun = entry?.grammar_noun;
  if (!noun) return null;

  const raw = typeof noun.gender === 'string' ? noun.gender.trim().toLowerCase() : '';
  if (ARTICLES.has(raw)) return raw;

  const nom = noun?.declension?.singular?.forms?.nominative;
  if (typeof nom === 'string') {
    const first = nom.trim().toLowerCase().split(/\s+/)[0];
    if (ARTICLES.has(first)) return first;
  }
  return null;
}

export function articleToGenderRO(article) {
  return GENDER_RO[article] ?? null;
}

/** academic_ro, else academic_en, else null. */
export function cleanTranslation(entry) {
  const ro = typeof entry?.academic_ro === 'string' ? entry.academic_ro.trim() : '';
  if (ro) return ro;
  const en = typeof entry?.academic_en === 'string' ? entry.academic_en.trim() : '';
  if (en) return en;
  return null;
}

/** Build a LessonWord, or null if the entry is not a usable noun. */
export function toLessonWord(entry, id) {
  const article = normalizeGender(entry);
  if (!article) return null;
  const translation = cleanTranslation(entry);
  if (!translation) return null;
  return {
    id,
    german: entry.word,
    article,
    translation,
    gender: articleToGenderRO(article),
    rank: entry.rank,
    ipa: typeof entry.ipa === 'string' ? entry.ipa : '',
  };
}
```

```js
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { normalizeGender, cleanTranslation, toLessonWord } from './lib/vocab-normalize.mjs';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = join(__dirname, '..');
const SOURCE = join(ROOT, 'data-source', 'enriched_vocab_full.json');
const ID_OFFSET = 100000; // keep generated ids clear of curated ids (1-28)

const raw = JSON.parse(readFileSync(SOURCE, 'utf-8'));
if (!Array.isArray(raw)) throw new Error('Expected source JSON to be an array');

const lessonWords = [];
const seenIds = new Set();
let nonNoun = 0;
let excludedNoGender = 0;
let excludedNoTranslation = 0;

for (const entry of raw) {
  if (!entry.grammar_noun) { nonNoun++; continue; }
  const id = ID_OFFSET + entry.rank;
  const lw = toLessonWord(entry, id);
  if (!lw) {
    if (!normalizeGender(entry)) excludedNoGender++;
    else if (!cleanTranslation(entry)) excludedNoTranslation++;
    continue;
  }
  if (seenIds.has(lw.id)) throw new Error(`Duplicate generated id ${lw.id} (${lw.german})`);
  if (lw.id <= 28) throw new Error(`Generated id ${lw.id} collides with curated range (1-28)`);
  seenIds.add(lw.id);
  lessonWords.push(lw);
}

lessonWords.sort((a, b) => a.rank - b.rank);
if (lessonWords.length === 0) throw new Error('No usable lesson words produced — aborting');

// Artifact 1: full dictionary (lazy-loaded from public/).
mkdirSync(join(ROOT, 'public', 'vocab'), { recursive: true });
writeFileSync(join(ROOT, 'public', 'vocab', 'dictionary.json'), JSON.stringify(raw));

// Artifact 2: bundled, type-checked lesson words.
mkdirSync(join(ROOT, 'src', 'data', 'generated'), { recursive: true });
const ts =
  `// AUTO-GENERATED by scripts/build-vocab.mjs — do not edit by hand.\n` +
  `import type { LessonWord } from '../vocab-types';\n\n` +
  `export const lessonWords: LessonWord[] = ${JSON.stringify(lessonWords)};\n`;
writeFileSync(join(ROOT, 'src', 'data', 'generated', 'lessonWords.ts'), ts);

console.log('--- build-vocab report ---');
console.log(`Total source entries:        ${raw.length}`);
console.log(`Non-noun (skipped):          ${nonNoun}`);
console.log(`Excluded — no valid gender:  ${excludedNoGender}`);
console.log(`Excluded — no translation:   ${excludedNoTranslation}`);
console.log(`Usable lesson words:         ${lessonWords.length}`);
console.log(`dictionary.json:             public/vocab/dictionary.json`);
console.log(`lessonWords.ts:              src/data/generated/lessonWords.ts`);
```

## Task 5: Dictionary loader

- [ ] **Step 1: Write the loader**

Create `src/data/dictionary.ts`:

```ts
import type { DictionaryEntry } from './vocab-types';

let cache: DictionaryEntry[] | null = null;
let inflight: Promise<DictionaryEntry[]> | null = null;

/** Fetch the full dictionary once and cache it in memory for the session. */
export async function loadDictionary(): Promise<DictionaryEntry[]> {
  if (cache) return cache;
  if (inflight) return inflight;
  inflight = fetch('/vocab/dictionary.json')
    .then((res) => {
      if (!res.ok) throw new Error(`Failed to load dictionary: ${res.status}`);
      return res.json() as Promise<DictionaryEntry[]>;
    })
    .then((data) => {
      cache = data;
      inflight = null;
      return data;
    })
    .catch((err) => {
      inflight = null;
      throw err;
    });
  return inflight;
}

/** Look up a single entry by its German headword (case-sensitive). */
export async function getEntry(german: string): Promise<DictionaryEntry | undefined> {
  const all = await loadDictionary();
  return all.find((e) => e.word === german);
}
```

```ts
const WORDS_PER_LESSON = 4;
const LESSONS_PER_UNIT = 5;

/**
 * Build successive frequency units from the remaining lesson nouns.
 * Each unit: up to 5 word-lessons (4 words each) + a review + a trophy,
 * with unlock requirements chained to the previous unit's trophy.
 */
function generateFrequencyUnits(startUnitNum: number, pool: LessonWord[]): Section[] {
  const sections: Section[] = [];
  const wordsPerUnit = WORDS_PER_LESSON * LESSONS_PER_UNIT;
  let unitNum = startUnitNum;

  for (let offset = 0; offset < pool.length; offset += wordsPerUnit) {
    const unitWords = pool.slice(offset, offset + wordsPerUnit);
    const nodes: LessonNode[] = [];

    for (let l = 0; l < LESSONS_PER_UNIT; l++) {
      const chunk = unitWords.slice(l * WORDS_PER_LESSON, (l + 1) * WORDS_PER_LESSON);
      if (chunk.length === 0) break;
      nodes.push({
        id: `u${unitNum}_l${l + 1}`,
        type: 'word',
        title: `Vocabular ${l + 1}`,
        items: chunk,
      });
    }

    nodes.push({ id: `u${unitNum}_review`, type: 'review', title: 'Recapitulare', items: [] });
    nodes.push({ id: `u${unitNum}_t1`, type: 'trophy', title: `Maestru U${unitNum}`, items: [] });

    const first = unitWords[0];
    const last = unitWords[unitWords.length - 1];
    sections.push({
      id: `sec${unitNum}`,
      title: `Unitatea ${unitNum}: Frecvență`,
      description: `Cuvinte frecvente (#${first.rank}–#${last.rank}).`,
      unlockRequirement: `u${unitNum - 1}_t1`,
      nodes,
    });
    unitNum++;
  }

  return sections;
}

export const learningPath: Section[] = [
  ...curatedPath,
  ...generateFrequencyUnits(26, lessonWords),
];
```
